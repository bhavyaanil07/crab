def search_local_pdfs(query: str) -> str:
    """Reads through your converted PDF data folder to extract matching content sentences."""
    combined_text = []
    if not os.path.exists(DATA_DIR):
        return "No local document directory found."
        
    pdf_files = [f for f in os.listdir(DATA_DIR) if f.endswith('.pdf')]
    if not pdf_files:
        return "No PDF documents available in the data folder."

    for file in pdf_files:
        try:
            reader = PdfReader(os.path.join(DATA_DIR, file))
            for page in reader.pages:
                text = page.extract_text()
                if text and any(word.lower() in text.lower() for word in query.split()):
                    combined_text.append(f"[From Document {file}]: {text[:800]}...")
        except Exception as e:
            continue
            
    return "\n\n".join(combined_text) if combined_text else "No specific matches located in local documentation files."

def search_web_or_amma(query: str, target_amma_site: bool = False) -> str:
    """Performs web search via Tavily, optionally forcing constraints to pull from amma.org."""
    try:
        search_query = f"site:amma.org {query}" if target_amma_site else query
        response = tavily_client.search(query=search_query, max_results=3)
        results = response.get("results", [])
        if results:
            return "\n".join([f"Source ({r['url']}): {r['content']}" for r in results])
    except Exception as e:
        return f"Search processing error: {str(e)}"
    return "No clear answers found on the web indexes."

def get_chatbot_response(user_msg):
    try:
        conversation_history.append({"role": "user", "content": user_msg})
        
        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=conversation_history,
            tools=tools,
            max_tokens=500,
            temperature=0.4
        )
        
        response_message = response.choices[0].message
        
        if response_message.tool_calls:
            conversation_history.append(response_message)
            
            for tool_call in response_message.tool_calls:
                func_name = tool_call.function.name
                args = json.loads(tool_call.function.arguments)
                
                if func_name == "search_local_pdfs":
                    tool_output = search_local_pdfs(args.get("query"))
                elif func_name == "search_web_or_amma":
                    tool_output = search_web_or_amma(args.get("query"), args.get("target_amma_site", False))
                else:
                    tool_output = "Unknown tool requested."
                    
                conversation_history.append({
                    "tool_call_id": tool_call.id,
                    "role": "tool",
                    "name": func_name,
                    "content": tool_output
                })
            
            second_response = client.chat.completions.create(
                model="gpt-4o-mini",
                messages=conversation_history,
                max_tokens=500,
                temperature=0.4
            )
            assistant_msg = second_response.choices[0].message.content
        else:
            assistant_msg = response_message.content
            
        conversation_history.append({"role": "assistant", "content": assistant_msg})
        return assistant_msg
        
    except Exception as e:
        print(f"Runtime Warning/Error: {e}")
        return "I'm sorry, I hit a technical snag cross-referencing my document sources."
